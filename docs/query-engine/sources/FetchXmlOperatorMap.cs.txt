using System;
using System.Collections.Generic;
using XrmGhost.FrameworkHost.QueryEngine.Abstractions;

namespace XrmGhost.FrameworkHost.QueryEngine.Frontends
{
    /// <summary>
    /// Maps a FetchXml condition operator string onto the <see cref="OperatorIdentity"/> used
    /// everywhere else in the QueryEngine, so a FetchXml query and the equivalent
    /// <c>QueryExpression</c> name the same operator with the same string.
    ///
    /// <para>
    /// Most FetchXml spellings differ from the SDK member name only by hyphens
    /// (<c>not-like</c> / <c>NotLike</c>, <c>on-or-before</c> / <c>OnOrBefore</c>), and
    /// <see cref="OperatorIdentity.Normalize"/> already reconciles those.  The table below carries
    /// <b>every</b> spelling normalisation cannot reach, and nothing else: the short comparison
    /// forms (<c>eq</c>, <c>ne</c>, <c>lt</c>, <c>le</c>, <c>gt</c>, <c>ge</c>), the negative forms
    /// FetchXml words differently from the SDK (<c>not-begin-with</c> / <c>DoesNotBeginWith</c>,
    /// <c>not-contain</c> / <c>DoesNotContain</c>), the spelled-out week
    /// (<c>last-seven-days</c> / <c>Last7Days</c>), the <c>eq-</c>/<c>ne-</c> forms of the
    /// contextual operators (<c>eq-userid</c> / <c>EqualUserId</c>,
    /// <c>eq-useroruserhierarchyandteams</c> / <c>EqualUserOrUserHierarchyAndTeams</c>) and the
    /// hierarchy forms that name equality first (<c>eq-or-under</c> / <c>UnderOrEqual</c>).
    /// <c>OperatorContractTests</c> walks the whole FetchXml spelling table against the SDK
    /// inventory, so an operator whose spelling this table fails to reach is a failing test rather
    /// than a query refused for the wrong reason.
    /// </para>
    ///
    /// <para>
    /// <b>And it carries what FetchXML cannot spell at all (ADO-2500).</b>  One operator of the
    /// pinned inventory has no FetchXML grammar: <c>MasksSelect</c> is accepted by the server
    /// through <c>QueryExpression</c>, with a bitwise semantics this engine implements, and refused
    /// under every FetchXML spelling — which makes it the first operator the two frontends
    /// legitimately disagree about, and the disagreement is the SERVER's rather than this engine's.
    /// <see cref="TryGetMissingFetchXmlGrammarReason"/> is that declaration, queryable on the
    /// canonical key and carrying the measurement behind it.  It is public because it is a
    /// <i>derived fact</i> that consumers outside this namespace need — the published support
    /// matrix among them — and the alternative is each of them keeping a roster of its own, which
    /// is the second-place-coverage-is-declared defect ADO-2286 removed from the registry.
    /// </para>
    ///
    /// <para>
    /// <b>Resolving is not supporting.</b> This type answers "which operator is this"; whether the
    /// engine can evaluate it is <see cref="IConditionEvaluatorRegistry.IsSupported"/>'s answer,
    /// and <see cref="IsSupported(string, IConditionEvaluatorRegistry)"/> forwards to it rather
    /// than asserting support from the presence of a mapping.  A key with no registered evaluator
    /// — the fiscal-calendar operators, the contextual ones, the hierarchy ones — resolves to its
    /// identity and is then rejected by the shared decision, never silently
    /// accepted.  The converse holds too: <c>between</c> and <c>contain-values</c> gained
    /// evaluators in ADO-2290, and the date operators in ADO-2291, without this table changing by
    /// a line, because support was never this type's answer to give.
    /// </para>
    /// </summary>
    public static class FetchXmlOperatorMap
    {
        // -----------------------------------------------------------------------
        // FetchXml-only spellings (normalised key → canonical identity)
        // -----------------------------------------------------------------------

        private static readonly Dictionary<string, string> _aliases =
            new Dictionary<string, string>(StringComparer.Ordinal)
            {
                // Short comparison forms
                { "eq",               "equal" },
                { "ne",               "notequal" },
                { "neq",              "notequal" },   // legacy alias
                { "lt",               "lessthan" },
                { "le",               "lessequal" },
                { "gt",               "greaterthan" },
                { "ge",               "greaterequal" },

                // Negative forms FetchXml words differently from the SDK member name.  The
                // separator-written SDK spellings — "does-not-begin-with", "contain-values",
                // "not-between", and every date keyword from "not-on" to "next-year" — need no
                // entry here: OperatorIdentity.Normalize already reaches them, which is why
                // ADO-2290 registered seven operators and ADO-2291 sixteen more without touching
                // this table.
                { "notbeginwith",     "doesnotbeginwith" },      // not-begin-with
                { "notendwith",       "doesnotendwith" },        // not-end-with
                { "notcontain",       "doesnotcontain" },        // not-contain
                { "notcontainvalues", "doesnotcontainvalues" },  // not-contain-values

                // The week FetchXml spells out and the SDK writes as a digit
                { "lastsevendays",    "last7days" },             // last-seven-days
                { "nextsevendays",    "next7days" },             // next-seven-days

                // Contextual operators: FetchXml abbreviates "Equal"/"NotEqual" to eq-/ne-
                { "equserid",                       "equaluserid" },                       // eq-userid
                { "neuserid",                       "notequaluserid" },                    // ne-userid
                { "eqbusinessid",                   "equalbusinessid" },                   // eq-businessid
                { "nebusinessid",                   "notequalbusinessid" },                // ne-businessid
                { "equserlanguage",                 "equaluserlanguage" },                 // eq-userlanguage
                { "equserteams",                    "equaluserteams" },                    // eq-userteams
                { "equseroruserteams",              "equaluseroruserteams" },              // eq-useroruserteams
                { "equseroruserhierarchy",          "equaluseroruserhierarchy" },          // eq-useroruserhierarchy
                { "equseroruserhierarchyandteams",  "equaluseroruserhierarchyandteams" },  // eq-useroruserhierarchyandteams

                // Hierarchy operators: FetchXml names equality first, the SDK last
                { "eqorunder",        "underorequal" },          // eq-or-under
                { "eqorabove",        "aboveorequal" },          // eq-or-above
            };

        // -----------------------------------------------------------------------
        // Operators with NO FetchXml grammar (canonical identity → measured reason)
        // -----------------------------------------------------------------------

        // The table above says which operator a FetchXml keyword names.  This one says which
        // operators FetchXml cannot name AT ALL, and it is a different claim: an entry here is not
        // a missing translation this engine could supply, it is a spelling the SERVER has no
        // grammar for, so translating it would build a plan Dataverse would never have accepted.
        //
        // Keyed on the CANONICAL identity and not on a spelling, which is the whole point: the
        // refusal below is decided after OperatorIdentity.Normalize has collapsed the separators
        // and the casing, so 'masks-select', 'masksselect', 'masks_select', 'MasksSelect' and
        // 'Masks-Select' are all refused by one entry — where a list of raw spellings would refuse
        // the ones somebody thought of and quietly translate the rest.
        //
        // The reason travels with the entry for the same reason OperatorSupport.Rejected demands
        // one: a refusal whose cause lives in a comment is, to the caller who reads a log,
        // indistinguishable from an operator nobody has got to yet.
        private static readonly Dictionary<string, string> _withoutFetchXmlGrammar =
            new Dictionary<string, string>(StringComparer.Ordinal)
            {
                {
                    "masksselect",
                    "MasksSelect has no FetchXML grammar: the Dataverse server answers every spelling " +
                    "tried — 'masks-select', 'masksselect', 'maskselect' and 'masks_select' — with HTTP " +
                    "400 'Unknown Condition Operator', measured through the FetchXML frontend " +
                    "(calibration/2026-09-17-mask-operators). The operator itself is NOT unsupported: " +
                    "the same server accepts ConditionOperator.MasksSelect through QueryExpression and " +
                    "answers it as '(value & operand) != 0' — four tables, and the semantics confirmed " +
                    "on three columns with zero counterexamples (ADO-2498, frozen in " +
                    "docs/query-engine/sources/2026-09-17-queryexpression-operatori-esito.md) — and this " +
                    "engine evaluates it on that path. What is missing is the FetchXML spelling, so a " +
                    "FetchXML query naming it is refused here rather than translated into a plan " +
                    "Dataverse would never have accepted."
                },
            };

        // -----------------------------------------------------------------------
        // Public API
        // -----------------------------------------------------------------------

        /// <summary>
        /// Resolves a FetchXml operator string to its canonical operator identity.
        /// </summary>
        /// <param name="fetchXmlOperator">
        /// The raw <c>operator</c> attribute value from a FetchXml <c>&lt;condition&gt;</c> element.
        /// </param>
        /// <returns>The canonical operator identity.</returns>
        /// <exception cref="ArgumentNullException">
        /// Thrown when <paramref name="fetchXmlOperator"/> is <c>null</c>.
        /// </exception>
        /// <exception cref="Exceptions.NotSupportedQueryOperatorException">
        /// Thrown when <paramref name="fetchXmlOperator"/> names no operator of the pinned SDK
        /// inventory.
        /// </exception>
        public static string Resolve(string fetchXmlOperator)
        {
            if (fetchXmlOperator == null) throw new ArgumentNullException(nameof(fetchXmlOperator));

            if (!TryResolve(fetchXmlOperator, out var canonical))
                throw new Exceptions.NotSupportedQueryOperatorException(fetchXmlOperator);

            return canonical;
        }

        /// <summary>
        /// Resolves a FetchXml operator string to its canonical operator identity.
        /// </summary>
        /// <returns>
        /// <c>false</c> when the string names no operator of the pinned SDK inventory.
        /// </returns>
        /// <exception cref="ArgumentNullException">
        /// Thrown when <paramref name="fetchXmlOperator"/> is <c>null</c>.
        /// </exception>
        public static bool TryResolve(string fetchXmlOperator, out string canonical)
        {
            if (fetchXmlOperator == null) throw new ArgumentNullException(nameof(fetchXmlOperator));

            string key = OperatorIdentity.Normalize(fetchXmlOperator);

            // The alias target is canonicalised too, so a typo in the table above cannot invent an
            // identity that no SDK operator carries.
            string candidate = _aliases.TryGetValue(key, out var aliased) ? aliased : key;

            if (!OperatorIdentity.TryCanonicalize(candidate, out canonical))
                return false;

            // An operator with no FetchXML grammar does not resolve FROM FETCHXML, whatever
            // spelling it was written in.  The test is made on the canonical identity the
            // normalised key resolves to — not on the raw string the caller wrote — so every
            // spelling OperatorIdentity.Normalize collapses onto 'masksselect' is refused by the
            // one entry, and a spelling nobody anticipated cannot slip past it.
            if (_withoutFetchXmlGrammar.ContainsKey(canonical))
            {
                canonical = null!;
                return false;
            }

            return true;
        }

        /// <summary>
        /// Whether <paramref name="canonicalKey"/> names an operator FetchXML has <b>no grammar
        /// for</b>, and if so why — the measurement behind the refusal, not a sentence composed at
        /// the call site.
        ///
        /// <para>
        /// This is the declaration <see cref="TryResolve"/> itself consults, exposed rather than
        /// duplicated: the caller that renders the refusal (<c>FetchXmlFrontend</c>) and the caller
        /// that publishes it read the same entry, so a query's refusal and the document describing
        /// it cannot disagree.  It is a <b>narrower</b> question than
        /// <see cref="IConditionEvaluatorRegistry.SupportOf"/>: an operator can be fully supported
        /// by this engine and still have no FetchXML spelling, which is exactly
        /// <c>masksselect</c>'s case, so a caller must not read a <c>true</c> here as "unsupported".
        /// </para>
        /// </summary>
        /// <param name="canonicalKey">
        /// The operator's canonical identity — what <see cref="OperatorIdentity.Of"/> returns.  A
        /// non-canonical spelling simply does not match: this is a lookup on the identity, and the
        /// spellings are <see cref="TryResolve"/>'s business.
        /// </param>
        /// <param name="reason">
        /// Why FetchXML cannot name the operator, or <c>null</c> when it can.
        /// </param>
        /// <exception cref="ArgumentNullException">
        /// Thrown when <paramref name="canonicalKey"/> is <c>null</c>.
        /// </exception>
        public static bool TryGetMissingFetchXmlGrammarReason(string canonicalKey, out string? reason)
        {
            if (canonicalKey == null) throw new ArgumentNullException(nameof(canonicalKey));

            return _withoutFetchXmlGrammar.TryGetValue(canonicalKey, out reason);
        }

        /// <summary>
        /// Returns <c>true</c> when the FetchXml operator both resolves to a canonical identity and
        /// has an evaluator registered in <paramref name="registry"/>.
        /// </summary>
        /// <exception cref="ArgumentNullException">
        /// Thrown when either argument is <c>null</c>.
        /// </exception>
        public static bool IsSupported(string fetchXmlOperator, IConditionEvaluatorRegistry registry)
        {
            if (fetchXmlOperator == null) throw new ArgumentNullException(nameof(fetchXmlOperator));
            if (registry == null) throw new ArgumentNullException(nameof(registry));

            return TryResolve(fetchXmlOperator, out var canonical) && registry.IsSupported(canonical);
        }
    }
}
